// Arduino：5 行核心，框架全包了
void setup() {
  pinMode(4, OUTPUT);          // 配置 = 一句话
}
void loop() {
  digitalWrite(4, HIGH);        // 出错？不存在的，它替你吞了
  delay(500);                  // 全局罚站式等待
  digitalWrite(4, LOW);
  delay(500);
}
// ESP-IDF：每个决定都要你来做
void app_main(void) {
  gpio_config_t cfg = {          // 配置 = 结构体摊开全部选项
    .pin_bit_mask = 1ULL << 4,
    .mode         = GPIO_MODE_OUTPUT,
  };
  ESP_ERROR_CHECK(gpio_config(&cfg)); // 出错？立即停机报告！
  while(1) {
    gpio_set_level(GPIO_NUM_4, 1);
    vTaskDelay(pdMS_TO_TICKS(500)); // 只睡本任务，别人照跑
    gpio_set_level(GPIO_NUM_4, 0);
    vTaskDelay(pdMS_TO_TICKS(500));
  }
}
